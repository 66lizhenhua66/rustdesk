cmake_minimum_required(VERSION 3.5.0)
project(harmony_controller)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(OHOS_ARCH STREQUAL "x86_64")
    set(RUST_TARGET "x86_64-unknown-linux-ohos")
    set(VPX_ABI "x86_64")
elseif(OHOS_ARCH STREQUAL "arm64-v8a" OR OHOS_ARCH STREQUAL "arm64")
    set(RUST_TARGET "aarch64-unknown-linux-ohos")
    set(VPX_ABI "arm64-v8a")
else()
    message(FATAL_ERROR "Unsupported OHOS_ARCH: ${OHOS_ARCH}")
endif()
set(RUST_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../../libs/controller-core")
set(RUST_LIB "${RUST_DIR}/target/${RUST_TARGET}/release/libremote_controller_core.a")
if(NOT EXISTS "${RUST_LIB}")
    message(FATAL_ERROR "Build controller-core first: ${RUST_LIB}")
endif()
set(VPX_ROOT "${RUST_DIR}/artifacts/libvpx-ohos/${VPX_ABI}")
set(VPX_LIB "${VPX_ROOT}/lib/libvpx.a")
if(NOT EXISTS "${VPX_LIB}")
    message(FATAL_ERROR "Build libvpx first: ${VPX_LIB}")
endif()
add_library(controller SHARED controller_napi.cpp video_renderer.cpp)
target_include_directories(controller PRIVATE "${RUST_DIR}/include" "${VPX_ROOT}/include")
target_link_libraries(controller PRIVATE "${RUST_LIB}" "${VPX_LIB}" libace_napi.z.so libnative_window.so libnative_buffer.so m)
